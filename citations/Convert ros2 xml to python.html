from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, ExecuteProcess
from launch.substitutions import LaunchConfiguration, Command, FindExecutable, PathJoinSubstitution
from launch_ros.actions import Node
from launch_ros.substitutions import FindPackageShare
from launch.conditions import IfCondition, UnlessCondition


def generate_launch_description():
    # Declare arguments
    use_jsp = LaunchConfiguration('use_jsp')

    declare_use_jsp = DeclareLaunchArgument(
        'use_jsp',
        default_value='gui',
        description="The joint state publisher to use: 'gui', 'jsp' [no gui, internal control], or 'none'."
    )

    # Define file paths
    turtle_brick_share = FindPackageShare('turtle_brick')
    urdf_file = PathJoinSubstitution([turtle_brick_share, 'turtle_bot.urdf.xacro'])
    rviz_file = PathJoinSubstitution([turtle_brick_share, 'peruse_turtle.rviz'])

    robot_state_publisher = Node(
        package='robot_state_publisher',
        executable='robot_state_publisher',
        parameters=[{

                urdf_file
            ])
        }]
    )

    # Joint State Publisher (GUI)
    joint_state_publisher_gui = Node(
        package='joint_state_publisher_gui',
        executable='joint_state_publisher_gui',
        condition=IfCondition("'"+use_jsp.perform(None)+"'=='gui'")
    )

    # Joint State Publisher (non-GUI)
    joint_state_publisher = Node(
        package='joint_state_publisher',
        executable='joint_state_publisher',
        condition=IfCondition("'"+use_jsp.perform(None)+"'=='jsp'")
    )

    # RViz (unless use_jsp == 'none')
    rviz = Node(
        package='rviz2',
        executable='rviz2',
        arguments=['-d', rviz_file],
        condition=UnlessCondition("'"+use_jsp.perform(None)+"'=='none'")
    )

    return LaunchDescription([
        declare_use_jsp,
        robot_state_publisher,
        joint_state_publisher_gui,
        joint_state_publisher,
        rviz

from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import XMLLaunchDescriptionSource
from launch_ros.actions import Node
from launch.substitutions import PathJoinSubstitution
from launch_ros.substitutions import FindPackageShare


def generate_launch_description():
    # Common package share directory
    turtle_brick_share = FindPackageShare('turtle_brick')

    # Paths
    show_turtle_launch = PathJoinSubstitution([turtle_brick_share, 'show_turtle.launch.xml'])
    turtle_yaml = PathJoinSubstitution([turtle_brick_share, 'turtle.yaml'])
    rviz_file = PathJoinSubstitution([turtle_brick_share, 'run_turtle.rviz'])

    # Include show_turtle.launch.xml with argument override
    show_turtle = IncludeLaunchDescription(
        XMLLaunchDescriptionSource(show_turtle_launch),
        launch_arguments={'use_jsp': 'none'}.items()
    )

    # Turtlesim node
    turtlesim_node = Node(
        package='turtlesim',
        executable='turtlesim_node',
        parameters=[{'holonomic': True}]
    )

    # Turtlebot node
    turtlebot_node = Node(
        package='turtle_brick',
        executable='turtlebot',
        parameters=[turtle_yaml],
        remappings=[
            ('pose', '/turtle1/pose'),
            ('cmd_vel', '/turtle1/cmd_vel')
        ]
    )

    # RViz node
    rviz = Node(
        package='rviz2',
        executable='rviz2',
        arguments=['-d', rviz_file]
    )

    # Return all actions
    return LaunchDescription([
        show_turtle,
        turtlesim_node,
        turtlebot_node,
        rviz
    ])

🧩 Explanation

  • IncludeLaunchDescription(XMLLaunchDescriptionSource(...)) replaces <include file="...">.

  • launch_arguments={'use_jsp': 'none'}.items() passes arguments into the included launch file.

  • PathJoinSubstitution and FindPackageShare replicate the $(find-pkg-share ...) syntax.

  • Parameters are passed as:

    • Dictionaries for inline key–value ({'holonomic': True})

    • YAML file paths for external configuration files.

  • remappings replaces <remap from="..." to="..."/>.

  • arguments=['-d', rviz_file] mirrors args="-d ...".

The <exec_depend> tag is not valid for the format version specified at the top of your package.xml.

🧾 Background

Each package.xml starts like this:

xml
<package format="N">

The schema for that format number defines which tags are allowed.

FormatIntroducedAllows <exec_depend>?
1ROS 1 (Hydro–Indigo)❌ No
2ROS 2 (Crystal+)✅ Yes
3ROS 2 newer (Foxy+)✅ Yes

So — if your package.xml starts like this:

xml
<package format="1">

then <exec_depend> is invalid; only <build_depend>, <run_depend>, etc., are allowed.

✅ How to fix it

Open your package.xml and change the first line from:

xml
<package format="1">

to:

xml
<package format="3">

(You can use format="2" if you’re on a very old ROS 2 release, but 3 is safest for Jazzy / Kilted / modern.)

That should make <exec_depend> valid and stop the schema validation failure.

🧠 What exec_depend means

<exec_depend> declares runtime-only dependencies — i.e. packages that your node needs when executed, not necessarily at build time.

Example:

xml
<exec_depend>rclpy</exec_depend>
<exec_depend>turtlesim</exec_depend>

These dependencies will be installed with rosdep install and checked during launch.

✅ In summary

  • Cause: <package format="1"> doesn’t support <exec_depend>.

  • Fix: Change it to <package format="3">.

  • Effect: Your manifest validates and colcon test will pass schema checking.

Even though format="3" does support <exec_depend>.
So the root problem must be that your XML doesn’t actually match the schema’s ordering rules.

🧩 The key detail

In package format 3, the XML schema requires elements to appear in a specific order — not arbitrary.
When tags appear out of order, the validator misreads the structure and flags the next valid tag as “unexpected.”

Here’s the correct order per the ROS 2 Package Manifest 3 spec:

  1. <name>

  2. <version>

  3. <description>

  4. <maintainer>

  5. <license>

  6. (optionally) <url>, <author>

  7. <buildtool_depend>

  8. <build_depend>

  9. <build_export_depend>

  10. <buildtool_export_depend>

  11. <exec_depend>

  12. <depend> (and the rest of the dependency tags)

  13. <test_depend>

  14. <doc_depend>

  15. <member_of_group>

  16. <export>

So:
➡️ You currently have <depend> after <exec_depend>, and <member_of_group> between dependencies and <export>.
This violates the expected order.

  <export>
    <build_type>ament_cmake</build_type>
  </export>
</package>

✅ Why this works

  • The tag ordering now matches the official package_format3.xsd.

  • <member_of_group> is valid after dependency declarations and before <export>.

  • <exec_depend> is recognized properly because the validator is no longer out of sequence.

from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import FrontendLaunchDescriptionSource
from launch_ros.actions import Node
from launch.substitutions import PathJoinSubstitution
from launch_ros.substitutions import FindPackageShare


def generate_launch_description():
    # Shared package path
    turtle_brick_share = FindPackageShare('turtle_brick')

    # File paths
    show_turtle_launch = PathJoinSubstitution([turtle_brick_share, 'show_turtle.launch.xml'])
    turtle_yaml = PathJoinSubstitution([turtle_brick_share, 'turtle.yaml'])
    rviz_file = PathJoinSubstitution([turtle_brick_share, 'run_turtle.rviz'])

    # Include the XML launch file (works with any frontend)
    show_turtle = IncludeLaunchDescription(
        FrontendLaunchDescriptionSource(show_turtle_launch),
        launch_arguments={'use_jsp': 'none'}.items()
    )

    # Turtlesim node
    turtlesim_node = Node(
        package='turtlesim',
        executable='turtlesim_node',
        parameters=[{'holonomic': True}]
    )

    # Turtlebot node
    turtlebot_node = Node(
        package='turtle_brick',
        executable='turtlebot',
        parameters=[turtle_yaml],
        remappings=[
            ('pose', '/turtle1/pose'),
            ('cmd_vel', '/turtle1/cmd_vel'),
        ]
    )

    # RViz node
    rviz_node = Node(
        package='rviz2',
        executable='rviz2',
        arguments=['-d', rviz_file]
    )

    return LaunchDescription([
        show_turtle,
        turtlesim_node,
        turtlebot_node,
        rviz_node,
    ])

🧩 Explanation

  • FrontendLaunchDescriptionSource
    Newer ROS 2 releases (Humble → Kilted) unified XMLLaunchDescriptionSource and YAMLLaunchDescriptionSource into this single class.
    It automatically detects XML, YAML, or Python frontends.

  • FindPackageShare + PathJoinSubstitution
    These replace $(find-pkg-share pkg)/file.

  • launch_arguments
    Replaces the <arg> tag inside your <include> block.

  • parameters=[turtle_yaml]
    Loads parameters from your YAML file.

  • remappings=[(...)]
    Mirrors your <remap> tags.

  • arguments=['-d', rviz_file]
    Matches args="-d ...", passing the RViz config file.

✅ Works on:

  • ROS 2 Humble

  • ROS 2 Iron

  • ROS 2 Jazzy

  • ROS 2 Kilted
